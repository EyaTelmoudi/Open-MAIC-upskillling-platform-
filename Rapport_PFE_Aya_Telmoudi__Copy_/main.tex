\makeatletter

\newlength{\chapter@number@width}

\def\@makechapterhead#1{%
  {%
  \normalfont
  \setlength{\parindent}{0pt}%
  \vspace*{10pt}%

  \settowidth{\chapter@number@width}{%
    \hbox{%
      \color{white}\Large\bfseries
      \hspace{\dimexpr 1mm+3pt}%
      \thechapter
      \hspace{\dimexpr 1mm+3pt}%
    }%
  }

  \hbox{

    % Bloc gauche
    \vtop{%
      \hsize=\dimexpr\chapter@number@width+\tabcolsep+2\fboxrule+\tabcolsep

      \begin{tabular}[t]{@{}c}
        \scshape
        \@chapapp

        \fboxsep=0pt
        \colorbox{black}{%
          \vbox{
            \hbox{\vbox to 0.7cm{}}
            \hbox{%
              \color{white}\Large\bfseries
              \thechapter
            }
            \hbox{\vbox to 5pt{}}
          }%
        }

      \end{tabular}%
    }%

    % Bloc droite
    \vtop{%
      \advance\hsize by -\dimexpr\chapter@number@width
      \hspace*{-0.4cm}

      \begin{tabular}[t]{c}
        \scshape

        \fboxsep=0pt
        \colorbox{white}{%
          \vbox{

            \hbox{\Large\bfseries #1}

          }%
        }

      \end{tabular}%
    }%
  }%

  \vspace{40pt}
  }%
}

\makeatother
